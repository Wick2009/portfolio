<script>
(function () {
  var STORAGE_KEY = "csa-dashboard-progress";
  var progress = {};
  try { progress = JSON.parse(localStorage.getItem(STORAGE_KEY)) || {}; } catch (e) { progress = {}; }

  var doneBoxes = document.querySelectorAll(".csa-done");
  var total = doneBoxes.length;

  function updateProgressLabel() {
    var done = 0;
    doneBoxes.forEach(function (cb) { if (cb.checked) done++; });
    document.getElementById("csa-progress").textContent = done + " / " + total + " lessons checked off";
  }

  doneBoxes.forEach(function (cb) {
    var key = cb.getAttribute("data-key");
    cb.checked = !!progress[key];
    cb.addEventListener("change", function () {
      progress[key] = cb.checked;
      localStorage.setItem(STORAGE_KEY, JSON.stringify(progress));
      updateProgressLabel();
    });
  });
  updateProgressLabel();

  var searchBox = document.getElementById("csa-search");
  var rows = document.querySelectorAll(".csa-row");

  function applyFilters() {
    var q = searchBox.value.trim().toLowerCase();
    rows.forEach(function (row) {
      var matchesSearch = !q || row.getAttribute("data-search").indexOf(q) !== -1;
      row.style.display = matchesSearch ? "" : "none";
    });
  }

  searchBox.addEventListener("input", applyFilters);
})();
</script>
